# مستند جامع پیاده‌سازی «کارگاه» برای Vibe Coding

> **دستورالعمل برای ابزار کدنویسی:** این سند یک specification کامل و بدون ابهام است. تمام تصمیمات فنی گرفته شده. هیچ پرسشی لازم نیست. هر بخش را به ترتیب پیاده‌سازی کن. هر جا «فاز ۱» نوشته شده، یعنی در MVP لازم است. «فاز ۲+» یعنی بعد از MVP.

## A.3. دستورات Bootstrap

```bash
# 1. ساخت monorepo
mkdir kargah && cd kargah
pnpm init
echo 'packages:\n  - "apps/*"\n  - "packages/*"' > pnpm-workspace.yaml

# 2. Next.js
pnpm dlx create-next-app@14.2.3 apps/web --typescript --tailwind --app --src-dir --import-alias "@/*" --no-eslint --use-pnpm

# 3. NestJS
pnpm dlx @nestjs/cli@10.3.8 new api --package-manager pnpm --skip-git
mv api apps/api

# 4. FastAPI
mkdir -p apps/nlp && cd apps/nlp
python3.11 -m venv venv
source venv/bin/activate
pip install fastapi==0.111.0 uvicorn==0.30.0 hazm==0.10.0 parsivar==0.2.4 fasttext==0.9.2 pydantic==2.7.1

# 5. Shared packages
mkdir -p packages/shared-types packages/shared-config
```

model ExerciseType {
  id                String           @id @default(uuid()) @db.Uuid
  slug              String           @unique @db.VarChar(100)
  title             String           @db.VarChar(200)
  category          ExerciseCategory
  accessLevel       UserRole         @default(NOVICE) @map("access_level")
  description       String           @db.Text
  durationSeconds   Int              @map("duration_seconds")
  wordTargetMin     Int              @map("word_target_min")
  wordTargetMax     Int              @map("word_target_max")
  bankSourceType    BankType?        @map("bank_source_type")
  rubricJson        Json             @default("{}") @map("rubric_json")
  skillWeightsJson  Json             @default("{}") @map("skill_weights_json")
  baseXp            Int              @default(30) @map("base_xp")
  difficultyBase    Float            @default(1.0) @map("difficulty_base")
  isActive          Boolean          @default(true) @map("is_active")
  displayOrder      Int              @default(0) @map("display_order")
  iconAsset         String?          @map("icon_asset")
  createdAt         DateTime         @default(now()) @map("created_at")
  updatedAt         DateTime         @updatedAt @map("updated_at")

  sessions          ExerciseSession[]
  recommendations   RecommendationQueue[]

  @@index([category, isActive])
  @@index([accessLevel])
  @@map("exercise_types")
}

model ExerciseSession {
  id                String        @id @default(uuid()) @db.Uuid
  userId            String        @map("user_id") @db.Uuid
  exerciseTypeId    String        @map("exercise_type_id") @db.Uuid
  status            SessionStatus @default(IN_PROGRESS)
  startedAt         DateTime      @default(now()) @map("started_at")
  endedAt           DateTime?     @map("ended_at")
  sourceItemId      String?       @map("source_item_id") @db.Uuid
  sourceItemType    BankType?     @map("source_item_type")
  promptSnapshotJson Json         @default("{}") @map("prompt_snapshot_json")
  rerollCount       Int           @default(0) @map("reroll_count")
  createdAt         DateTime      @default(now()) @map("created_at")

  user              User          @relation(fields: [userId], references: [id])
  exerciseType      ExerciseType  @relation(fields: [exerciseTypeId], references: [id])
  drafts            Draft[]
  submission        Submission?
  events            EventLog[]

  @@index([userId, status])
  @@index([exerciseTypeId])
  @@index([startedAt])
  @@map("exercise_sessions")
}

model Draft {
  id             String          @id @default(uuid()) @db.Uuid
  sessionId      String          @map("session_id") @db.Uuid
  userId         String          @map("user_id") @db.Uuid
  contentText    String          @map("content_text") @db.Text
  wordCount      Int             @default(0) @map("word_count")
  versionNumber  Int             @map("version_number")
  saveTrigger    DraftSaveTrigger @map("save_trigger")
  createdAt      DateTime        @default(now()) @map("created_at")

  session        ExerciseSession @relation(fields: [sessionId], references: [id])
  user           User            @relation(fields: [userId], references: [id])

  @@unique([sessionId, versionNumber])
  @@index([userId, createdAt])
  @@map("drafts")

**کد کلیدی `ShuffleBagSelector`:**

```typescript
async select(input: SelectorInput): Promise<SelectorOutput> {
  const { userId, bankType, filters } = input;

  let cycle = await this.prisma.userCycle.findFirst({
    where: { userId, bankType, isActive: true }
  });

  if (!cycle) {
    cycle = await this.startNewCycle(userId, bankType, filters);
  }

  const itemIds: string[] = cycle.itemIdsJson as string[];
  const usedIds: string[] = cycle.usedItemIdsJson as string[];
  const remaining = itemIds.filter(id => !usedIds.includes(id));

  const candidates = await this.filterCandidates(remaining, filters);

  if (candidates.length === 0) {
    const relaxed = await this.relaxFilters(remaining, filters);
    if (relaxed.length === 0) {
      cycle = await this.startNewCycle(userId, bankType, filters);
      return this.select({ ...input, filters: {} });
    }
  }

  const weights = await this.computeWeights(candidates, userId, bankType);
  const seed = this.makeSeed(userId, bankType, new Date().toDateString());
  const rng = seededRandom(seed);

  let bestKey = -1;
  let selected: string | null = null;
  for (const [id, w] of Object.entries(weights)) {
    const clamped = Math.min(w, Math.max(...Object.values(weights)) / 4);
    const r = rng();
    const key = Math.pow(r, 1 / clamped);
    if (key > bestKey) { bestKey = key; selected = id; }
  }

  const positionInCycle = `${usedIds.length + 1}/${itemIds.length}`;
  const log = await this.prisma.selectionLog.create({
    data: {
      userId, bankType, itemId: selected!,
      algorithm: 'SHUFFLE_BAG',
      cycleId: cycle.id,
      cyclePosition: positionInCycle,
      weightsSnapshotJson: weights,
      filtersJson: filters ?? {},
      alternativesCount: candidates.length - 1,
      seed
    }
  });

  await this.prisma.userCycle.update({
    where: { id: cycle.id },
    data: { usedItemIdsJson: [...usedIds, selected!] }
  });

  if (usedIds.length + 1 >= itemIds.length) {
    await this.prisma.userCycle.update({
      where: { id: cycle.id },
      data: { isActive: false, completedAt: new Date() }
    });
  }

  await this.eventsService.log(userId, 'bank_item_selected', {
    bankType, itemId: selected, cycleId: cycle.id,
    positionInCycle, algorithm: 'SHUFFLE_BAG'
  });

  const item = await this.fetchItem(bankType, selected!);
  return { item, selectionLogId: log.id, cyclePosition: positionInCycle };
}
```

## E.4. ماژول Integrity

## E.5. ماژول Gamification — XP

**کد محاسبه XP:**

```typescript
async calculateXp(submission: Submission, exerciseType: ExerciseType, userStats: UserStats): Promise<XpResult> {
  const base = exerciseType.baseXp;

  const durationFactor = Math.min(submission.durationSeconds / exerciseType.durationSeconds, 1.5);

  const completionFactor = submission.integrityResult === 'PASSED' ? 1.0
                          : submission.integrityResult === 'WARNING' ? 0.8
                          : 0.3;

  const recentSameType = await this.prisma.eventLog.count({
    where: {
      userId: submission.userId,
      eventName: 'exercise_submitted',
      eventPayloadJson: { path: ['exerciseTypeId'], equals: exerciseType.id },
      serverTimestamp: { gte: subDays(new Date(), 7) }
    }
  });
  const noveltyMultiplier = recentSameType > 0 ? 0.7 : 1.0;

  const qualityMultiplier = 1.0;

  const yesterdayActive = await this.prisma.streakDay.findFirst({
    where: { userId: submission.userId, date: startOfDay(subDays(new Date(), 1)), status: 'ACTIVE' }
  });
  const consistencyBonus = yesterdayActive ? 1.1 : 1.0;

  const difficultyFactor = exerciseType.difficultyBase;

  const priorSessions = await this.prisma.exerciseSession.count({
    where: { userId: submission.userId, exerciseTypeId: exerciseType.id, status: 'SUBMITTED' }
  });
  const firstTimeFactor = priorSessions <= 1 ? 1.5 : 1.0;

  const last24h = await this.prisma.eventLog.count({
    where: {
      userId: submission.userId,
      eventName: 'exercise_submitted',
      eventPayloadJson: { path: ['exerciseTypeId'], equals: exerciseType.id },
      serverTimestamp: { gte: subHours(new Date(), 24) }
    }
  });
  const diminishing = 1 / (1 + 0.3 * Math.max(0, last24h - 1));

  const todayXp = await this.prisma.xPLedger.aggregate({
    where: { userId: submission.userId, createdAt: { gte: startOfDay(new Date()) } },
    _sum: { finalXp: true }
  });
  const todaySum = todayXp._sum.finalXp ?? 0;
  const dailyCapFactor = todaySum >= 1000 ? 0 : todaySum >= 600 ? 0.5 : 1.0;

  const raw = base * durationFactor * completionFactor * difficultyFactor * firstTimeFactor;
  const finalXp = Math.round(raw * noveltyMultiplier * qualityMultiplier * consistencyBonus * diminishing * dailyCapFactor);

  return {
    base, finalXp,
    multipliers: { durationFactor, completionFactor, noveltyMultiplier, qualityMultiplier, consistencyBonus, difficultyFactor, firstTimeFactor, diminishing, dailyCapFactor }
  };
}
```

**سطح:**

```typescript
function levelFromXp(xp: number): number {
  if (xp < 100) return 1;
  const n = Math.floor((-80 + Math.sqrt(6400 + 80 * (100 + xp))) / 40);
  return Math.max(1, n + 1);
}
```

async evaluate(rule: BadgeRule, event: EventLog): Promise<boolean> {
  const ctx = await this.buildContext(rule, event);
  const passed = await this.evaluateCondition(rule.conditionDslJson as Condition, ctx);

  if (!passed) {

  }

  if (rule.verificationLevel === 'A_AUTOMATIC' && rule.verificationQueryJson) {
    const verified = await this.runVerificationQuery(rule, event.userId);
    if (!verified) {
      await this.eventsService.log(event.userId, 'badge_verification_failed', { ruleId: rule.id });
      return false;
    }
  }

  await this.awardBadge(rule, event);
  return true;
}

const BASE = '/api/v1';

class ApiError extends Error {
  constructor(public status: number, public code: string, message: string, public details?: any) {

async function request<T>(path: string, init: RequestInit = {}): Promise<T> {
  const token = typeof window !== 'undefined' ? localStorage.getItem('token') : null;
  const res = await fetch(`${BASE}${path}`, {
    ...init,
    headers: {
      'Content-Type': 'application/json',
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
      ...init.headers,
    },
  });

  if (!res.ok) {
    const body = await res.json().catch(() => ({}));
    if (res.status === 401) {
      localStorage.removeItem('token');
      if (typeof window !== 'undefined') window.location.href = '/auth/login';
    }
    throw new ApiError(res.status, body.error ?? 'UNKNOWN', body.message ?? 'خطا', body.details);
  }
  return res.json();
}

function weightedRandomWithoutReplacement<T>(
  items: { item: T; weight: number }[],
  k: number = 1,
  rng: () => number
): T[] {
  const keyed = items.map(({ item, weight }) => ({
    item,
    key: Math.pow(rng(), 1 / weight)
  }));
  keyed.sort((a, b) => b.key - a.key);
  return keyed.slice(0, k).map(k => k.item);
}

import { createHmac } from 'crypto';

function makeSeed(userId: string, bankType: string, dateStr: string, secret: string): string {
  return createHmac('sha256', secret).update(`${userId}|${bankType}|${dateStr}`).digest('hex');

function seededRandom(seed: string): () => number {
  let a = parseInt(seed.slice(0, 8), 16);
  return function() {
    a |= 0; a = a + 0x6D2B79F5 | 0;
    let t = Math.imul(a ^ a >>> 15, 1 | a);
    t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;

function checkLipogram(text: string, forbiddenLetter: string): { violation: boolean; offendingWords: string[] } {
  const normalized = normalizePersian(text);
  const forbidden = normalizePersian(forbiddenLetter);

این سند تمام تصمیمات فنی، اسکیماها، endpoints، کامپوننت‌ها، الگوریتم‌ها، داده‌های اولیه، و ترتیب پیاده‌سازی را مشخص کرده. **هیچ تصمیم باقی‌مانده‌ای وجود ندارد.** ابزار Vibe Coding می‌تواند بدون هیچ سؤالی از گام ۱ شروع کند.

**اصل راهنما:** هر تصمیمی که در این سند نیامده، به سمت **سادگی، سازگاری با RTL، و «نوشتن در جلو، ثبت در پشت»** حل می‌شود. اگر باز هم ابهامی بود، **ساده‌ترین گزینه‌ای که با فاز ۱ سازگار است** انتخاب می‌شود و در کد با کامنت `// DECISION: ...` ثبت می‌شود.
